### 2.4.1 Component Structure

```typescript
// components/EmployeeForm/EmployeeForm.tsx
import { useState } from 'react';
import type { EmployeeCreateRequest } from '@/types/employee';

interface EmployeeFormProps {
  onSubmit: (data: EmployeeCreateRequest) => Promise<void>;
  isLoading?: boolean;
}

export function EmployeeForm({ onSubmit, isLoading = false }: EmployeeFormProps) {
  const [fullName, setFullName] = useState('');
  const [email, setEmail] = useState('');
  const [error, setError] = useState<string | null>(null);

  const handleSubmit = async (e: React.FormEvent) => {
    e.preventDefault();
    setError(null);
    try {
      await onSubmit({ full_name: fullName, email });
    } catch (err) {
      setError(err instanceof Error ? err.message : 'Unknown error');
    }
  };

  return (
    <form onSubmit={handleSubmit} aria-label="Employee form">
      <label htmlFor="fullName">Full Name</label>
      <input
        id="fullName"
        value={fullName}
        onChange={(e) => setFullName(e.target.value)}
        required
        aria-invalid={!!error}
      />
      <label htmlFor="email">Email</label>
      <input
        id="email"
        type="email"
        value={email}
        onChange={(e) => setEmail(e.target.value)}
        required
      />
      {error && <p role="alert">{error}</p>}
      <button type="submit" disabled={isLoading}>
        {isLoading ? 'Saving...' : 'Save'}
      </button>
    </form>
  );
}
```

### 2.5.2 ตัวอย่าง Service (Clean Architecture)

```python
# pkg/employee/service.py
from datetime import datetime
from pkg.employee.models import Employee, EmployeeCreate, EmployeeDTO
from pkg.employee.repository import EmployeeRepository


class EmailTakenError(Exception): pass
class ValidationError(Exception): pass


class EmployeeService:
    def __init__(self, repo: EmployeeRepository) -> None:
        self.repo = repo

    async def create(self, payload: EmployeeCreate) -> EmployeeDTO:
        if not payload.full_name or not payload.full_name.strip():
            raise ValidationError("full_name is required")
        if await self.repo.find_by_email(payload.email) is not None:
            raise EmailTakenError("email already exists")

        now = datetime.utcnow()
        employee = Employee(
            full_name=payload.full_name.strip(),
            email=payload.email,
            created_at=now,
            updated_at=now,
        )
        saved = await self.repo.save(employee)
        return EmployeeDTO(**saved.model_dump())
```

## 3.6 ตัวอย่าง Integration Design (Oracle ERP)

```python
# pkg/integration/oracle_erp/client.py
import httpx
from tenacity import retry, stop_after_attempt, wait_exponential


class OracleERPClient:
    def __init__(self, base_url: str, token: str) -> None:
        self.base_url = base_url
        self.token = token

    @retry(
        stop=stop_after_attempt(3),
        wait=wait_exponential(multiplier=1, min=2, max=10),
    )
    async def post_payment(self, payload: dict, idempotency_key: str) -> dict:
        async with httpx.AsyncClient(timeout=30) as client:
            resp = await client.post(
                f"{self.base_url}/payments",
                json=payload,
                headers={
                    "Authorization": f"Bearer {self.token}",
                    "Idempotency-Key": idempotency_key,
                },
            )
            resp.raise_for_status()
            return resp.json()
```

```python
from httpx import AsyncClient, ASGITransport
from api.main import app


@pytest.mark.asyncio
async def test_create_employee_success():
    transport = ASGITransport(app=app)
    async with AsyncClient(transport=transport, base_url="http://test") as client:
        resp = await client.post(
            "/employee",
            json={"full_name": "Somchai", "email": "somchai@example.com"},
        )
        assert resp.status_code == 201
        data = resp.json()
        assert data["full_name"] == "Somchai"
        assert data["created_at"] == data["updated_at"]


@pytest.mark.asyncio
async def test_create_employee_duplicate_email():
    transport = ASGITransport(app=app)
    async with AsyncClient(transport=transport, base_url="http://test") as client:
        await client.post(
            "/employee",
            json={"full_name": "A", "email": "dup@example.com"},
        )
        resp = await client.post(
            "/employee",
            json={"full_name": "B", "email": "dup@example.com"},
        )
        assert resp.status_code == 409
```

## 5.4 GitHub Actions Workflow (ตัวอย่าง)

```yaml
# .github/workflows/ci-cd.yml
name: CI/CD Pipeline

on:
  push:
    branches: [main, develop, 'release/**']
  pull_request:
    branches: [main, develop]

jobs:
  ci:
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: actions/setup-python@v5
        with:
          python-version: '3.11'
      - name: Install deps
        run: pip install -r requirements.txt
      - name: Lint
        run: ruff check .
      - name: Type check
        run: mypy pkg/
      - name: Test
        run: pytest --cov=pkg --cov-fail-under=80
      - name: Security scan
        uses: aquasecurity/trivy-action@master
        with:
          scan-type: 'fs'
          severity: 'CRITICAL,HIGH'

  build:
    needs: ci
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - uses: docker/build-push-action@v5
        with:
          push: true
          tags: your-org/pms-api:${{ github.sha }}

  deploy-dev:
    needs: build
    if: github.ref == 'refs/heads/develop'
    runs-on: ubuntu-latest
    environment: development
    steps:
      - name: Deploy to Dev
        run: kubectl set image deployment/pms-api pms-api=your-org/pms-api:${{ github.sha }} -n dev

  deploy-staging:
    needs: build
    if: startsWith(github.ref, 'refs/heads/release/')
    runs-on: ubuntu-latest
    environment: staging
    steps:
      - name: Deploy to Staging
        run: kubectl set image deployment/pms-api pms-api=your-org/pms-api:${{ github.sha }} -n staging

  deploy-prod:
    needs: build
    if: github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    environment:
      name: production
    steps:
      - name: Blue-Green Deploy
        run: ./scripts/deploy-blue-green.sh ${{ github.sha }}
```

## 6.2 Code Review Workflow

```
Code Assistant (Dev) ──► PR ──► CI Pipeline ──► Code Review ──► Update ──► Deploy
                                    │              │
                                    ▼              ▼
                              ❌ Fail → Fix    ❌ Request Changes → Fix
                                              ✅ Approve → Merge
```

## 6.3 Code Review Criteria (10 ข้อ)

### หมวดที่ 1: ความถูกต้อง (Does it work?)

1. **ตรงตาม Requirement** — ทำงานถูกต้องตาม Ticket/User Story
2. **รองรับ Edge Cases** — Null, Empty, Emoji, Network Loss
3. **ความปลอดภัย** — Input Validation, ไม่มี SQL Injection/XSS/Hardcoded Secrets

### หมวดที่ 2: คุณภาพโค้ด (Is it clean?)

4. **อ่านง่ายและสื่อความหมาย** — ชื่อตัวแปร/ฟังก์ชันชัดเจน
5. **ไม่ทำงานซ้ำซ้อน (DRY)** — ยุบรวมเป็น Function กลาง
6. **ประสิทธิภาพ** — ไม่มี N+1 Query, ไม่มี Loop ซ้อนโดยไม่จำเป็น
7. **สไตล์และมาตรฐาน** — ตรงตาม Linter และ Architecture

### หมวดที่ 3: การทดสอบและการดูแลรักษา (Can we maintain it?)

8. **การทดสอบ** — Unit Test ครอบคลุม, Arrange-Act-Assert, Isolation
9. **Observability** — Structured Logging, Metric, Trace
10. **Documentation** — API Contract, ADR, README อัปเดต

## 8.4 การแก้ไขปัญหาเบื้องต้น

| Export ไม่ได้ | Browser Block | อนุญาต Pop-up |

```yaml
name: Database Migration
on:
  push:
    branches: [develop, main, 'release/**']
    paths: ['database/migrations/**']

jobs:
  migrate-dev:
    if: github.ref == 'refs/heads/develop'
    runs-on: ubuntu-latest
    steps:
      - uses: actions/checkout@v4
      - name: Run migrations
        run: |
          docker run --rm \
            -e DATABASE_URL=${{ secrets.DEV_DATABASE_URL }} \
            migrate/migrate -path=/migrations -database ${{ secrets.DEV_DATABASE_URL }} up

  migrate-prod:
    if: github.ref == 'refs/heads/main'
    runs-on: ubuntu-latest
    environment: production
    steps:
      - uses: actions/checkout@v4
      - name: Backup
        run: ./scripts/backup-database.sh production
      - name: Dry-run

      - name: Wait for approval
        uses: trstringer/manual-approval@v1
        with:
          approvers: senior-dev-team
          minimum-approvals: 2
      - name: Run migrations
        run: ./scripts/run-migration.sh production
```

# 📘 เล่ม 13: คู่มือ RCA (Root Cause Analysis)

## 13.1 RCA คืออะไร

**RCA (Root Cause Analysis)** = การวิเคราะห์หาสาเหตุที่แท้จริงของปัญหา เพื่อแก้ที่ต้นเหตุไม่ให้เกิดซ้ำ

## 13.2 ขั้นตอนการทำ RCA (6 ขั้น)

### ขั้นที่ 1: ระบุปัญหา (Define the Problem)
- เกิดอะไรขึ้น?
- เกิดขึ้นเมื่อไหร่?
- ใครได้รับผลกระทบ?
- ผลกระทบรุนแรงแค่ไหน?

### ขั้นที่ 2: รวบรวมข้อมูล (Collect Data)
- Log, Metric, Trace
- Timeline ของเหตุการณ์
- Monitoring/Alerting
- สัมภาษณ์ผู้เกี่ยวข้อง

### ขั้นที่ 3: ระบุสาเหตุที่เป็นไปได้ (Identify Possible Causes)
- ใช้ **5 Whys** หรือ **Fishbone Diagram**
- ระดมสมอง
- จัดกลุ่ม (Man, Machine, Method, Material, Measurement, Environment)

### ขั้นที่ 4: หา "สาเหตุหลัก" (Find the Root Cause)
- วิเคราะห์ลงลึก
- ตรวจสอบด้วยหลักฐาน
- ยืนยันด้วยการทดสอบ

### ขั้นที่ 5: วางแผนและแก้ไข (Implement Solution)
- Corrective Action (แก้ไข)
- Preventive Action (ป้องกัน)
- กำหนด Owner + Timeline

### ขั้นที่ 6: ติดตามผล (Monitor)
- ตรวจสอบว่าปัญหาไม่กลับมา
- วัดผลด้วย KPI
- อัปเดต Knowledge Base

```python
        if user.role not in roles:
            raise HTTPException(status_code=403, detail="Forbidden")
        return user
    return checker
```

## 14.7 Incident Response (Security)

```
Detect ──► Contain ──► Eradicate ──► Recover ──► Lessons Learned
```

- **Detect:** SIEM, Alert, User Report
- **Contain:** Isolate System, Revoke Credentials
- **Eradicate:** Remove Threat, Patch
- **Recover:** Restore, Monitor
- **Lessons Learned:** RCA + Preventive Action
